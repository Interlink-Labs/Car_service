                                                                                from typing import List, Optional
from fastapi import FastAPI, HTTPException, status
from pydantic import BaseModel

app = FastAPI(
    title="Автосервис - Модуль 'Детали'", 
    version="1.0.0",
    description="API для управления каталогом деталей и проверки их совместимости"
)

# --- Схемы данных (Pydantic) ---

class DetailBase(BaseModel):
    name: str
    article: str
    price: float
    compatible_cars: List[str]  # Список моделей машин, к которым подходит деталь

class DetailCreate(DetailBase):
    pass

class DetailResponse(DetailBase):
    id: int

# --- Имитация базы данных ---
db_details = [
    {"id": 1, "name": "Тормозные колодки передние", "article": "BR-101", "price": 2500.0, "compatible_cars": ["Focus", "Solaris"]},
    {"id": 2, "name": "Фильтр масляный", "article": "OF-99", "price": 600.0, "compatible_cars": ["Focus", "Rio", "Logan"]},
]
id_counter = 3


# --- Эндпоинты (Маршруты) ---

# 1. Добавить деталь (POST /api/v1/detail)
@app.post("/api/v1/detail", response_model=DetailResponse, status_code=status.HTTP_201_CREATED)
def create_detail(detail: DetailCreate):
    global id_counter
    new_detail = detail.dict()
    new_detail["id"] = id_counter
    db_details.append(new_detail)
    id_counter += 1
    return new_detail

# 2. Читать одну деталь по ID (GET /api/v1/detail/{detail_id})
@app.get("/api/v1/detail/{detail_id}", response_model=DetailResponse)
def get_detail(detail_id: int):
    for detail in db_details:
        if detail["id"] == detail_id:
            return detail
    raise HTTPException(status_code=404, detail="Деталь не найдена")

# 3. Обновить деталь (PUT /api/v1/detail/{detail_id})
@app.put("/api/v1/detail/{detail_id}", response_model=DetailResponse)
def update_detail(detail_id: int, updated_detail: DetailCreate):
    for detail in db_details:
        if detail["id"] == detail_id:
            detail.update(updated_detail.dict())
            return detail
    raise HTTPException(status_code=404, detail="Деталь не найдена")

# 4. Удалить деталь (DELETE /api/v1/detail/{detail_id})
@app.delete("/api/v1/detail/{detail_id}", status_code=status.HTTP_204_NO_CONTENT)
def delete_detail(detail_id: int):
    for index, detail in enumerate(db_details):
        if detail["id"] == detail_id:
            db_details.pop(index)
            return
    raise HTTPException(status_code=404, detail="Деталь не найдена")

# 5. Список деталей (GET /api/v1/list_details)
@app.get("/api/v1/list_details", response_model=List[DetailResponse])
def get_all_details():
    return db_details

# 6. Проверить совместимость (GET /api/v1/detail/check)
@app.get("/api/v1/detail/check")
def check_compatibility(article: str, car_model: str):
    for detail in db_details:
        if detail["article"].lower() == article.lower():
            is_compatible = car_model.lower() in [car.lower() for car in detail["compatible_cars"]]
            return {
                "article": article,
                "car_model": car_model,
                "compatible": is_compatible,
                "message": "Деталь подходит" if is_compatible else "Деталь не совместима с указанным автомобилем"
            }
    raise HTTPException(status_code=404, detail="Деталь с таким артикулом не найдена")
